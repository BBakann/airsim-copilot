**Goal:** Add a separate MJPEG video channel (front + 3rd-person top camera) so the mobile Canlı screen shows smooth video with a Ön/Üst toggle and tap-to-fullscreen — without touching autonomy, telemetry, copilot, estop, or MCP.

**Architecture:** Backend gets an isolated `VideoState` + `/video/push` (sim writes) + `/video` (MJPEG `multipart/x-mixed-replace`, mobile reads). A new sim `VideoPusher` daemon thread captures both cameras and pushes JPEGs ~12 FPS, independent of the autonomy/telemetry loop. Mobile renders the MJPEG endpoint in a WebView (RN `<Image>` cannot play multipart) with a view toggle and a fullscreen Modal; the existing JPEG-poll camera stays as fallback.

**Tech Stack:** Python 3.10 (Flask streaming, opencv, requests, pytest), AirSim, Expo SDK 54 + `react-native-webview`, jest+ts-jest. `PYTHONIOENCODING=utf-8`.

```python
# backend/video_state.py
from __future__ import annotations

import threading
import time
from typing import Optional


class VideoState:
    """Thread-safe son-kare deposu. view -> (bytes, ts). state.py'den ayrı."""

    def __init__(self, stale_s: float = 2.0) -> None:
        self._lock = threading.Lock()
        self._frames: dict[str, tuple[bytes, float]] = {}
        self._stale_s = stale_s

    def set_frame(self, view: str, data: bytes) -> None:
        with self._lock:
            self._frames[view] = (data, time.time())

    def get_frame(self, view: str) -> Optional[tuple[bytes, float]]:
        with self._lock:
            return self._frames.get(view)

    def is_stale(self, view: str, now: Optional[float] = None) -> bool:
        v = self.get_frame(view)
        if v is None:
            return True
        now = time.time() if now is None else now
        return now - v[1] > self._stale_s
```

```python
vstate = VideoState()

_VIEWS = ("front", "top")
# 1x1 gri JPEG (frame yokken placeholder)
_PLACEHOLDER_JPEG = bytes.fromhex(
    "ffd8ffe000104a46494600010100000100010000ffdb004300"
    "08060607060508070707090909"
    "0c140d0c0b0b0c1912130f141d1a1f1e1d1a1c1c20242e2720"
    "222c231c1c2837292c30313434341f27393d38323c2e333432"
    "ffc0000b080001000101011100ffc40014000100000000000000"
    "00000000000000000009ffc40014100100000000000000000000"
    "00000000000000ffda0008010100003f00bf80ffd9"
)


def _mjpeg_chunk(data: bytes) -> bytes:
    return (b"--frame\r\n"
            b"Content-Type: image/jpeg\r\n"
            b"Content-Length: " + str(len(data)).encode() + b"\r\n\r\n"
            + data + b"\r\n")


def frame_iter(store, view, max_frames=None, sleep_s=1.0 / 12):
    """Bounded test edilebilir MJPEG generator. max_frames=None -> sonsuz."""
    import time as _t
    n = 0
    while max_frames is None or n < max_frames:
        v = store.get_frame(view)
        if v is not None and not store.is_stale(view):
            yield _mjpeg_chunk(v[0])
        else:
            yield _mjpeg_chunk(_PLACEHOLDER_JPEG)
        n += 1
        if max_frames is None or n < max_frames:
            _t.sleep(sleep_s)
```

```python
# video_pusher.py
from __future__ import annotations

import threading
import time
from typing import Any, Optional

import cv2
import numpy as np

VIDEO_FPS = 12
JPEG_QUALITY = 80


def encode_jpeg(img_bgr, quality: int = JPEG_QUALITY) -> bytes:
    ok, buf = cv2.imencode(".jpg", img_bgr,
                           [cv2.IMWRITE_JPEG_QUALITY, int(quality)])
    if not ok:
        return b""
    return buf.tobytes()


def _resp_to_bgr(resp) -> Optional["np.ndarray"]:
    try:
        arr = np.frombuffer(resp.image_data_uint8, dtype=np.uint8)
        return arr.reshape(resp.height, resp.width, 3)
    except Exception:
        return None


def capture_payloads(client: Any) -> dict[str, bytes]:
    """AirSim'den ön("0") + üst("top") kare -> {view: jpeg bytes}."""
    import airsim
    reqs = [
        airsim.ImageRequest("0", airsim.ImageType.Scene, False, False),
        airsim.ImageRequest("top", airsim.ImageType.Scene, False, False),
    ]
    responses = client.simGetImages(reqs)
    out: dict[str, bytes] = {}
    for view, resp in zip(("front", "top"), responses):
        img = _resp_to_bgr(resp)
        if img is not None:
            jpg = encode_jpeg(img)
            if jpg:
                out[view] = jpg
    return out


def push_once(base: str, client: Any, http: Any, timeout: float = 0.5) -> None:
    try:
        payloads = capture_payloads(client)
    except Exception:
        return
    for view, data in payloads.items():
        try:
            http.post(f"{base}/video/push?view={view}", data=data,
                      timeout=timeout,
                      headers={"Content-Type": "application/octet-stream"})
        except Exception:
            pass


class VideoPusher:
    """Otonomiden bağımsız daemon: ~VIDEO_FPS kare push. Sürüşü etkilemez."""

    def __init__(self, client: Any, base_url: str, http: Any = None,
                 fps: int = VIDEO_FPS) -> None:
        self._client = client
        self._base = base_url.rstrip("/")
        self._period = 1.0 / max(1, fps)
        self._stop = threading.Event()
        self._thread: Optional[threading.Thread] = None
        if http is not None:
            self._http = http
        else:
            import requests
            self._http = requests.Session()

    def _run(self) -> None:
        while not self._stop.is_set():
            push_once(self._base, self._client, self._http)
            self._stop.wait(self._period)

    def start(self) -> None:
        if self._thread is None:
            self._thread = threading.Thread(target=self._run, daemon=True)
            self._thread.start()

    def stop(self) -> None:
        self._stop.set()
        if self._thread is not None:
            self._thread.join(timeout=2.0)
```

```tsx
import React, { useState } from "react";
import { ActivityIndicator, Image, Modal, Pressable, Text, View } from "react-native";

import VideoStream from "./VideoStream";
import { VideoView, videoUrl } from "../../utils/videoUrl";

interface Props {
  imageUrl?: string | null;
  isLoading?: boolean;
  isConnected?: boolean;
}

const raw = process.env.EXPO_PUBLIC_API_BASE_URL;
const BASE = (raw && raw.replace(/\/$/, "")) || "";

function Inner({
  view,
  setView,
  imageUrl,
  isLoading,
  isConnected,
}: {
  view: VideoView;
  setView: (v: VideoView) => void;
  imageUrl?: string | null;
  isLoading?: boolean;
  isConnected?: boolean;
}) {
  const url = videoUrl(BASE, view);
  return (
    <>
      {url ? (
        <VideoStream url={url} view={view} />
      ) : imageUrl ? (
        <Image
          source={{ uri: `data:image/jpeg;base64,${imageUrl}` }}
          className="h-full w-full"
          resizeMode="cover"
        />
      ) : (
        <View className="h-full w-full items-center justify-center bg-[#070b14]">
          {isLoading ? (
            <ActivityIndicator color="#38bdf8" />
          ) : (
            <Text className="text-slate-500">Kamera bekleniyor…</Text>
          )}
        </View>
      )}

      {/* HUD üst */}
      <View className="absolute left-3 right-3 top-3 flex-row justify-between">
        <View className="flex-row items-center gap-1 rounded-full bg-black/55 px-3 py-1">
          <View
            className={`h-2 w-2 rounded-full ${
              isConnected ? "bg-emerald-400" : "bg-red-500"
            }`}
          />
          <Text className="text-[11px] font-semibold text-white">
            {isConnected ? "CANLI" : "ÇEVRİMDIŞI"}
          </Text>
        </View>
        <View className="flex-row overflow-hidden rounded-full bg-black/55">
          {(["front", "top"] as VideoView[]).map((v) => (
            <Pressable
              key={v}
              onPress={() => setView(v)}
              className={`px-3 py-1 ${view === v ? "bg-sky-500" : ""}`}
            >
              <Text className="text-[11px] font-semibold text-white">
                {v === "front" ? "Ön" : "Üst"}
              </Text>
            </Pressable>
          ))}
        </View>
      </View>
    </>
  );
}

export default function RobotCameraView({
  imageUrl,
  isLoading,
  isConnected,
}: Props) {
  const [view, setView] = useState<VideoView>("front");
  const [full, setFull] = useState(false);

  return (
    <>
      <Pressable
        onPress={() => setFull(true)}
        className="relative w-full overflow-hidden rounded-3xl border border-[#2563eb55] bg-black aspect-[3/4]"
      >
        <Inner
          view={view}
          setView={setView}
          imageUrl={imageUrl}
          isLoading={isLoading}
          isConnected={isConnected}
        />
      </Pressable>

      <Modal visible={full} animationType="fade" onRequestClose={() => setFull(false)}>
        <View className="flex-1 bg-black">
          <Inner
            view={view}
            setView={setView}
            imageUrl={imageUrl}
            isLoading={isLoading}
            isConnected={isConnected}
          />
          <Pressable
            onPress={() => setFull(false)}
            className="absolute bottom-6 self-center rounded-full bg-white/20 px-6 py-3"
          >
            <Text className="text-base font-bold text-white">Kapat</Text>
          </Pressable>
        </View>
      </Modal>
    </>
  );
}
```
